def get_video_resolution(ts_file_path):
    try:
        # Open the TS file
        cap = cv2.VideoCapture(ts_file_path)
        # Get the resolution
        width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        return width, height
    except Exception as e:
        print(f"Error: {e}")
        return None
