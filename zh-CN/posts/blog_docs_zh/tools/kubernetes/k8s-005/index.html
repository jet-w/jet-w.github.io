第 5 章：配置管理与存储

学习目标
  • 掌握ConfigMap和Secret的使用
  • 理解PersistentVolume和PersistentVolumeClaim
  • 学会配置不同类型的存储后端
  • 熟练进行应用配置的动态更新

知识点

为什么需要配置管理

在容器化应用中，将配置与代码分离是最佳实践。Kubernetes提供了ConfigMap和Secret来管理配置数据。

ConfigMap

ConfigMap用于存储非敏感的配置数据。

创建ConfigMap

# 方式1：从字面值创建
kubectl create configmap app-config \
  --from-literal=APP_ENV=production \
  --from-literal=APP_DEBUG=false \
  --from-literal=LOG_LEVEL=info

# 方式2：从文件创建
echo "server.port=8080
server.host=0.0.0.0
database.pool.size=10" > app.properties

kubectl create configmap app-config-file --from-file=app.properties

# 方式3：从目录创建（包含多个文件）
mkdir config
echo "database_url=postgres://localhost/mydb" > config/database.conf
echo "redis_url=redis://localhost:6379" > config/cache.conf
kubectl create configmap app-config-dir --from-file=config/

# 查看创建的 ConfigMap
kubectl get configmap
kubectl describe configmap app-config
kubectl get configmap app-config -o yaml

YAML方式创建ConfigMap

apiVersion: v1
kind: ConfigMap
metadata:
  name: app-config
data:
  # 简单键值对
  APP_ENV: "production"
  APP_DEBUG: "false"
  LOG_LEVEL: "info"

  # 多行配置文件
  app.properties: |
    server.port=8080
    server.host=0.0.0.0
    database.pool.size=10

  # JSON配置
  config.json: |
    {
      "database": {
        "host": "localhost",
        "port": 5432
      },
      "cache": {
        "enabled": true,
        "ttl": 3600
      }
    }

  # Nginx配置示例
  nginx.conf: |
    server {
        listen 80;
        server_name localhost;

        location / {
            root /usr/share/nginx/html;
            index index.html;
        }

        location /api {
            proxy_pass http://backend:8080;
        }
    }

使用ConfigMap

作为环境变量

apiVersion: v1
kind: Pod
metadata:
  name: config-env-pod
spec:
  containers:
  - name: app
    image: busybox
    command: ["sh", "-c", "env && sleep 3600"]
    env:
    # 引用单个键
    - name: APP_ENVIRONMENT
      valueFrom:
        configMapKeyRef:
          name: app-config
          key: APP_ENV
    - name: DEBUG_MODE
      valueFrom:
        configMapKeyRef:
          name: app-config
          key: APP_DEBUG
          optional: true  # 键不存在时不报错
    # 引用所有键
    envFrom:
    - configMapRef:
        name: app-config
        optional: false

作为Volume挂载

apiVersion: v1
kind: Pod
metadata:
  name: config-volume-pod
spec:
  containers:
  - name: app
    image: nginx
    volumeMounts:
    - name: config-volume
      mountPath: /etc/nginx/conf.d
      readOnly: true
    - name: app-config-volume
      mountPath: /app/config
  volumes:
  # 挂载整个 ConfigMap
  - name: config-volume
    configMap:
      name: nginx-config
  # 挂载特定键
  - name: app-config-volume
    configMap:
      name: app-config
      items:
      - key: config.json
        path: settings.json  # 重命名文件
      - key: app.properties
        path: app.properties
        mode: 0644  # 设置文件权限

子路径挂载（不覆盖目录）

apiVersion: v1
kind: Pod
metadata:
  name: subpath-pod
spec:
  containers:
  - name: nginx
    image: nginx
    volumeMounts:
    # 使用 subPath 只挂载单个文件，不覆盖整个目录
    - name: config-volume
      mountPath: /etc/nginx/nginx.conf
      subPath: nginx.conf
  volumes:
  - name: config-volume
    configMap:
      name: nginx-config

ConfigMap热更新

apiVersion: apps/v1
kind: Deployment
metadata:
  name: config-reload-demo
spec:
  replicas: 1
  selector:
    matchLabels:
      app: demo
  template:
    metadata:
      labels:
        app: demo
      annotations:
        # 通过annotation触发重新部署
        configHash: "{{ .Values.configHash }}"
    spec:
      containers:
      - name: app
        image: nginx
        volumeMounts:
        - name: config
          mountPath: /etc/nginx/conf.d
      volumes:
      - name: config
        configMap:
          name: nginx-config
# 更新 ConfigMap
kubectl edit configmap app-config

# 或使用 kubectl apply
kubectl apply -f updated-configmap.yaml

# 注意：Volume挂载的ConfigMap会自动更新（可能有延迟）
# 环境变量不会自动更新，需要重启Pod

# 强制重启 Deployment 以应用新配置
kubectl rollout restart deployment config-reload-demo

Secret

Secret用于存储敏感数据，如密码、令牌、证书等。

Secret类型

🔄 正在渲染 Mermaid 图表...
flowchart TB
    subgraph SecretTypes["Secret类型"]
        Opaque["Opaque<br/>通用类型"]
        TLS["kubernetes.io/tls<br/>TLS证书"]
        DockerConfig["kubernetes.io/dockerconfigjson<br/>Docker仓库认证"]
        BasicAuth["kubernetes.io/basic-auth<br/>基本认证"]

        Token["kubernetes.io/service-account-token<br/>ServiceAccount令牌"]
    end

创建Secret

# 方式1：从字面值创建（Opaque类型）
kubectl create secret generic db-secret \
  --from-literal=username=admin \
  --from-literal=password='S3cr3t!Pass'

# 方式2：从文件创建
echo -n 'admin' > ./username.txt
echo -n 'S3cr3t!Pass' > ./password.txt
kubectl create secret generic db-secret-file \
  --from-file=username=./username.txt \
  --from-file=password=./password.txt

# 方式3：创建TLS Secret
kubectl create secret tls tls-secret \
  --cert=path/to/tls.crt \
  --key=path/to/tls.key

# 方式4：创建Docker仓库认证Secret
kubectl create secret docker-registry my-registry \
  --docker-server=registry.example.com \
  --docker-username=user \
  --docker-password=password \
  --docker-email=user@example.com

# 查看Secret
kubectl get secrets
kubectl describe secret db-secret
# 注意：data内容是base64编码的

# 解码查看Secret值
kubectl get secret db-secret -o jsonpath='{.data.password}' | base64 -d

YAML方式创建Secret

apiVersion: v1
kind: Secret
metadata:
  name: db-credentials
type: Opaque
# data中的值必须是base64编码
data:
  username: YWRtaW4=          # echo -n 'admin' | base64
  password: UzNjcjN0IVBhc3M=  # echo -n 'S3cr3t!Pass' | base64

---
# 使用stringData可以直接写明文（会自动转为base64）
apiVersion: v1
kind: Secret
metadata:
  name: db-credentials-plain
type: Opaque
stringData:
  username: admin
  password: "S3cr3t!Pass"

---
# TLS Secret
apiVersion: v1
kind: Secret
metadata:
  name: tls-secret
type: kubernetes.io/tls
data:
  tls.crt: <base64-encoded-cert>
  tls.key: <base64-encoded-key>

使用Secret

apiVersion: v1
kind: Pod
metadata:
  name: secret-demo-pod
spec:
  containers:
  - name: app
    image: nginx
    env:
    # 方式1：引用单个键
    - name: DB_USERNAME
      valueFrom:
        secretKeyRef:
          name: db-credentials
          key: username
    - name: DB_PASSWORD
      valueFrom:
        secretKeyRef:
          name: db-credentials
          key: password
    # 方式2：引用所有键
    envFrom:
    - secretRef:
        name: db-credentials
        optional: false
    # 方式3：Volume挂载
    volumeMounts:
    - name: secret-volume
      mountPath: /etc/secrets
      readOnly: true
  volumes:
  - name: secret-volume
    secret:
      secretName: db-credentials
      defaultMode: 0400  # 设置文件权限

使用Docker仓库Secret

apiVersion: v1
kind: Pod
metadata:
  name: private-image-pod
spec:
  containers:
  - name: app
    image: registry.example.com/my-app:v1.0
  imagePullSecrets:
  - name: my-registry

---
# 或者在 ServiceAccount 中配置
apiVersion: v1
kind: ServiceAccount
metadata:
  name: my-service-account
imagePullSecrets:
- name: my-registry
Secret安全注意事项
  1. Secret默认以Base64编码存储，并非加密，任何有权限的用户都可以解码
  2. 建议启用etcd加密（Encryption at Rest）
  3. 使用RBAC限制Secret的访问权限
  4. 考虑使用外部密钥管理系统（Vault、AWS Secrets Manager等）
  5. 不要将Secret提交到版本控制系统

Volume类型

# emptyDir - 临时存储，Pod删除时数据丢失
apiVersion: v1
kind: Pod
metadata:
  name: emptydir-pod
spec:
  containers:
  - name: writer
    image: busybox
    command: ["sh", "-c", "echo 'Hello' > /data/message && sleep 3600"]
    volumeMounts:
    - name: shared-data
      mountPath: /data
  - name: reader
    image: busybox
    command: ["sh", "-c", "cat /data/message && sleep 3600"]
    volumeMounts:
    - name: shared-data
      mountPath: /data
  volumes:
  - name: shared-data
    emptyDir: {}
    # 或使用内存存储
    # emptyDir:
    #   medium: Memory
    #   sizeLimit: 100Mi

---
# hostPath - 挂载宿主机目录
apiVersion: v1
kind: Pod
metadata:
  name: hostpath-pod
spec:
  containers:
  - name: app
    image: nginx
    volumeMounts:
    - name: host-data
      mountPath: /data
  volumes:
  - name: host-data
    hostPath:
      path: /var/data
      type: DirectoryOrCreate  # Directory, File, Socket, etc.

PersistentVolume (PV)

PV是集群中的一块存储，由管理员提前创建或通过StorageClass动态供应。

# 静态创建 PV
apiVersion: v1
kind: PersistentVolume
metadata:
  name: pv-nfs
  labels:
    type: nfs
spec:
  capacity:
    storage: 10Gi
  volumeMode: Filesystem
  accessModes:
  - ReadWriteMany      # RWX: 多节点读写
  persistentVolumeReclaimPolicy: Retain  # Retain/Delete/Recycle
  storageClassName: nfs-storage
  nfs:
    server: 192.168.1.100
    path: /exports/data

---
# 本地存储 PV
apiVersion: v1
kind: PersistentVolume
metadata:
  name: pv-local
spec:
  capacity:
    storage: 100Gi
  volumeMode: Filesystem
  accessModes:
  - ReadWriteOnce      # RWO: 单节点读写
  persistentVolumeReclaimPolicy: Delete
  storageClassName: local-storage
  local:
    path: /mnt/disks/ssd1
  nodeAffinity:
    required:
      nodeSelectorTerms:
      - matchExpressions:
        - key: kubernetes.io/hostname
          operator: In
          values:
          - node-1

---
# AWS EBS PV
apiVersion: v1
kind: PersistentVolume
metadata:
  name: pv-ebs
spec:
  capacity:
    storage: 50Gi
  accessModes:
  - ReadWriteOnce

PersistentVolumeClaim (PVC)

PVC是用户对存储的请求，Kubernetes会自动绑定合适的PV。

apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: my-pvc
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 5Gi
  storageClassName: nfs-storage
  # 可选：选择特定的PV
  selector:
    matchLabels:
      type: nfs

---
# 在Pod中使用PVC
apiVersion: v1
kind: Pod
metadata:
  name: pvc-pod
spec:
  containers:
  - name: app
    image: nginx
    volumeMounts:
    - name: data
      mountPath: /var/www/html
  volumes:
  - name: data
    persistentVolumeClaim:
      claimName: my-pvc

访问模式说明

访问模式简写说明
ReadWriteOnceRWO单节点读写
ReadOnlyManyROX多节点只读
ReadWriteManyRWX多节点读写
ReadWriteOncePodRWOP单Pod读写（K8s 1.22+）

StorageClass

StorageClass定义了存储类型，支持动态创建PV。

# NFS StorageClass（需要NFS Provisioner）
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: nfs-client
provisioner: nfs-subdir-external-provisioner
parameters:
  archiveOnDelete: "false"
reclaimPolicy: Delete
volumeBindingMode: Immediate
allowVolumeExpansion: true

---
# AWS EBS StorageClass
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: fast-ebs
provisioner: kubernetes.io/aws-ebs
parameters:
  type: gp3
  iops: "3000"
  throughput: "125"
  fsType: ext4
  encrypted: "true"
reclaimPolicy: Delete
volumeBindingMode: WaitForFirstConsumer
allowVolumeExpansion: true

---
# 本地存储 StorageClass
apiVersion: storage.k8s.io/v1
kind: StorageClass
metadata:
  name: local-storage
provisioner: kubernetes.io/no-provisioner
volumeBindingMode: WaitForFirstConsumer
# 查看 StorageClass
kubectl get storageclass

# 设置默认 StorageClass
kubectl patch storageclass nfs-client \
  -p '{"metadata": {"annotations":{"storageclass.kubernetes.io/is-default-class":"true"}}}'

# 使用默认StorageClass的PVC（不指定storageClassName）
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: dynamic-pvc
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 10Gi
  # 不指定storageClassName将使用默认StorageClass
  # 或明确指定
  storageClassName: fast-ebs

扩展PVC容量

# 确保 StorageClass 支持扩展
kubectl get storageclass fast-ebs -o yaml | grep allowVolumeExpansion

# 编辑 PVC 增加容量
kubectl patch pvc my-pvc -p '{"spec":{"resources":{"requests":{"storage":"20Gi"}}}}'

# 查看扩展状态
kubectl get pvc my-pvc
kubectl describe pvc my-pvc

StatefulSet存储

StatefulSet为每个Pod提供稳定的持久化存储。

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: mysql
spec:
  serviceName: mysql
  replicas: 3
  selector:
    matchLabels:
      app: mysql
  template:
    metadata:
      labels:
        app: mysql
    spec:
      containers:
      - name: mysql
        image: mysql:8.0
        ports:
        - containerPort: 3306
        env:
        - name: MYSQL_ROOT_PASSWORD
          valueFrom:
            secretKeyRef:
              name: mysql-secret
              key: root-password
        volumeMounts:
        - name: data
          mountPath: /var/lib/mysql
  volumeClaimTemplates:
  - metadata:
      name: data
    spec:
      accessModes:
      - ReadWriteOnce
      storageClassName: fast-ebs
      resources:
        requests:
          storage: 50Gi
# StatefulSet会为每个Pod创建独立的PVC
kubectl get pvc
# NAME           STATUS   VOLUME                                     CAPACITY
# data-mysql-0   Bound    pvc-xxx-0                                  50Gi
# data-mysql-1   Bound    pvc-xxx-1                                  50Gi
# data-mysql-2   Bound    pvc-xxx-2                                  50Gi

# 删除StatefulSet时，PVC不会自动删除
kubectl delete statefulset mysql
kubectl get pvc  # PVC仍然存在

# 需要手动删除PVC
kubectl delete pvc -l app=mysql

常见问题排查

PVC状态Pending

# 查看PVC状态
kubectl describe pvc my-pvc

# 常见原因：
# 1. 没有匹配的PV
kubectl get pv
# 检查容量、访问模式、StorageClass是否匹配

# 2. StorageClass不存在或配置错误
kubectl get storageclass
kubectl describe storageclass <name>

# 3. 存储后端问题
# 检查Provisioner日志
kubectl logs -n kube-system -l app=nfs-provisioner

Secret/ConfigMap更新后不生效

# 检查Pod使用方式
# 环境变量方式：需要重启Pod
kubectl rollout restart deployment my-app

# Volume方式：会自动更新，但有延迟（最多1分钟）
# 使用subPath挂载时不会自动更新！

# 强制触发更新
kubectl patch deployment my-app \
  -p "{\"spec\":{\"template\":{\"metadata\":{\"annotations\":{\"date\":\"$(date +%s)\"}}}}}"

# 检查存储类型和IOPS
kubectl get storageclass -o yaml

kubectl describe pv <pv-name>
