apiVersion: argoproj.io/v1alpha1
kind: Rollout
metadata:
  name: blue-green-demo
spec:
  replicas: 3
  revisionHistoryLimit: 2
  selector:
    matchLabels:
      app: blue-green-demo
  template:
    metadata:
      labels:
        app: blue-green-demo
    spec:
      containers:
      - name: app
        image: nginx:1.24
        ports:
        - containerPort: 80
        resources:
          requests:
            cpu: 100m
            memory: 128Mi
          limits:
            cpu: 200m
            memory: 256Mi
  strategy:
    blueGreen:
      # 正式服务（生产流量）
      activeService: blue-green-active
      # 预览服务（用于测试新版本）
      previewService: blue-green-preview
      # 自动升级（设为false需要手动确认）
      autoPromotionEnabled: false
      # 切换后保留旧版本的时间
      scaleDownDelaySeconds: 30
      # 预览版本就绪后等待时间
      previewReplicaCount: 1
---

kubectl argo rollouts get rollout blue-green-demo

# 更新镜像触发部署
kubectl argo rollouts set image blue-green-demo app=nginx:1.25

# 查看部署进度（实时监控）
kubectl argo rollouts get rollout blue-green-demo --watch

kubectl argo rollouts promote blue-green-demo

kubectl argo rollouts abort blue-green-demo
