Argo Events 事件驱动

第7章：使用 Argo Events 构建事件驱动架构

Argo Events 是一个事件驱动的工作流自动化框架，可以将各种事件源（如 Webhook、消息队列、定时器等）与 Kubernetes 资源或 Argo Workflows 连接起来。

7.1.3 安装 Argo Events

# 创建命名空间
kubectl create namespace argo-events

# 安装 Argo Events
kubectl apply -n argo-events -f https://raw.githubusercontent.com/argoproj/argo-events/stable/manifests/install.yaml

# 安装 EventBus（使用 NATS）
kubectl apply -n argo-events -f https://raw.githubusercontent.com/argoproj/argo-events/stable/manifests/install-validating-webhook.yaml

# 验证安装
kubectl get pods -n argo-events

7.2.2 NATS EventBus

# eventbus-nats.yaml
apiVersion: argoproj.io/v1alpha1
kind: EventBus
metadata:
  name: default
  namespace: argo-events
spec:
  nats:
    native:
      # 副本数
      replicas: 3
      # 认证策略
      auth: token
      # 持久化配置
      persistence:
        storageClassName: standard
        accessMode: ReadWriteOnce
        volumeSize: 10Gi
      # 容器资源配置
      containerTemplate:
        resources:
          requests:
            cpu: 100m
            memory: 128Mi
          limits:
            cpu: 500m
            memory: 512Mi
      # 指标配置
      metricsContainerTemplate:
        resources:
          requests:
            cpu: 50m
            memory: 64Mi
      # 反亲和性
      affinity:
        podAntiAffinity:
          preferredDuringSchedulingIgnoredDuringExecution:
          - weight: 100
            podAffinityTerm:
              labelSelector:
                matchLabels:
                  controller: eventbus-controller
              topologyKey: kubernetes.io/hostname

  name: default
  namespace: argo-events
spec:
  kafka:
    url: kafka-cluster.kafka:9092
    topic: argo-events
    version: "3.4.0"
    # TLS 配置
    tls:
      caCertSecret:
        name: kafka-ca
        key: ca.crt
      clientCertSecret:
        name: kafka-client-cert
        key: tls.crt
      clientKeySecret:
        name: kafka-client-key
        key: tls.key
    # SASL 认证
    sasl:
      mechanism: SCRAM-SHA-512
      userSecret:
        name: kafka-credentials
        key: username
      passwordSecret:
        name: kafka-credentials
        key: password
    # 消费者组配置
    consumerGroup:
      groupName: argo-events-sensors
      rebalanceStrategy: sticky

7.3 EventSource 事件源

7.3.1 Webhook EventSource

# eventsource-webhook.yaml
apiVersion: argoproj.io/v1alpha1
kind: EventSource
metadata:
  name: webhook
  namespace: argo-events
spec:
  service:
    ports:
    - port: 12000
      targetPort: 12000
  webhook:
    # 通用 webhook
    generic:
      port: "12000"
      endpoint: /generic
      method: POST
    # CI/CD webhook
    cicd:
      port: "12000"
      endpoint: /cicd
      method: POST
      # 过滤器
      filter:
        expression: "body.action == 'deploy'"
    # 带认证的 webhook
    secure:
      port: "12000"
      endpoint: /secure
      method: POST
      # 使用 HMAC 认证
      authSecret:
        name: webhook-secret
        key: token
---
# 暴露 webhook 服务
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: webhook-ingress
  namespace: argo-events
spec:
  rules:
  - host: events.example.com
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: webhook-eventsource-svc
            port:
              number: 12000

7.3.2 GitHub EventSource

# eventsource-github.yaml
apiVersion: argoproj.io/v1alpha1
kind: EventSource
metadata:
  name: github
  namespace: argo-events
spec:
  service:
    ports:
    - port: 12000
      targetPort: 12000
  github:
    # 监听特定仓库
    app-repo:
      # 仓库信息
      repositories:
      - owner: myorg
        names:
        - myapp
        - mylib
      # 监听的事件类型
      events:
      - push
      - pull_request
      - release
      # Webhook 配置
      webhook:
        endpoint: /github/app
        port: "12000"
        method: POST
      # GitHub API 配置
      apiToken:
        name: github-access
        key: token
      # Webhook Secret
      webhookSecret:
        name: github-webhook-secret
        key: secret
      # 是否验证 SSL
      insecure: false
      # 活跃状态
      active: true
      # 内容类型
      contentType: json
    # 监听组织级别事件
    org-events:
      owner: myorg
      events:
      - repository
      - member
      - team
      webhook:
        endpoint: /github/org
        port: "12000"
        method: POST
      apiToken:
        name: github-access
        key: token

      insecure: true

7.3.6 Calendar/Cron EventSource

# eventsource-calendar.yaml
apiVersion: argoproj.io/v1alpha1
kind: EventSource
metadata:
  name: calendar
  namespace: argo-events
spec:
  calendar:
    # 每日报告
    daily-report:
      # Cron 表达式
      schedule: "0 8 * * *"  # 每天早上8点
      # 时区
      timezone: "Asia/Shanghai"
      # 元数据
      metadata:
        task: daily-report
        priority: high
    # 每小时任务
    hourly-cleanup:
      schedule: "0 * * * *"  # 每小时
      timezone: "UTC"
      metadata:
        task: cleanup
    # 工作日任务
    workday-sync:
      schedule: "30 9 * * 1-5"  # 工作日9:30
      timezone: "Asia/Shanghai"
      metadata:
        task: data-sync
    # 使用间隔
    interval-task:
      interval: 30m  # 每30分钟
      metadata:
        task: health-check

# eventsource-redis.yaml
